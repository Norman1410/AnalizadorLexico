| =========================================
| Calculadora simple para probar el compilador
| - get: lee int/float
| - show: imprime string/int/float/bool
| - decide of: selección de operación
| - loop + exit when: repetir hasta salir
| =========================================

world string banner = "=== CALCULADORA ===" endl


coal navidad ¿ ? ¡
    local int op = 0 endl
    local float a = 0.0 endl
    local float b = 0.0 endl
    local float res = 0.0 endl

    show banner endl

    loop
        show "Menu: 1=+, 2=-, 3=*, 4=/, 0=salir" endl
        show "Operacion:" endl
        get op endl

        decide of
            op != 0 -> ¡
                show "a (float):" endl
                get a endl
                
                show "b (float):" endl
                get b endl
                show "Debug b:" endl
                show b endl


                decide of
                    op == 1 -> ¡
                        res = a + b endl
                        show "a + b =" endl
                        show res endl
                    !
                    op == 2 -> ¡
                        res = a - b endl
                        show "a - b =" endl
                        show res endl
                    !
                    op == 3 -> ¡
                        res = a * b endl
                        show "a * b =" endl
                        show res endl
                    !
                    op == 4 -> ¡
                        decide of
                            b != 0.0 -> ¡
                                res = a / b endl
                                show "a / b =" endl
                                show res endl
                            !
                            else -> ¡
                                show "Error: division por cero" endl
                            !
                        end decide endl
                    !
                    else -> ¡
                        show "Opcion invalida" endl
                    !
                end decide endl

            !
            else -> ¡
                | Si op == 0, salimos
            !
        end decide endl

        exit when op == 0 endl

    end loop endl
!
